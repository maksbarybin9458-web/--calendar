    <script>
        // Сюда будут загружаться смены с сервера
        let userShifts = {}; 
        let currentDate = new Date();
        let selectedUser = "";

        // Ссылка на твой будущий веб-сервер (скрипт Гугла)
        const API_URL = "ТВОЯ_ССЫЛКА_НА_СКРИПТ_ГУГЛА";

        async function login() {
            const input = document.getElementById("userInput").value.trim();
            if (!input) return alert("Введи данные!");
            
            selectedUser = input.toLowerCase();
            
            // Показываем загрузку
            document.getElementById("authScreen").innerHTML = "<h2>Загрузка графика...</h2>";
            
            try {
                // Стучимся в Гугл Таблицу за данными конкретного человека
                const response = await fetch(`${API_URL}?user=${encodeURIComponent(selectedUser)}`);
                userShifts = await response.json();
                
                document.getElementById("authScreen").style.display = "none";
                document.getElementById("appScreen").style.display = "flex";
                
                renderCalendar();
            } catch (e) {
                alert("Ошибка загрузки. Проверь сеть или имя. Пока загрузим демо-режим.");
                // ДЕМО-ДАННЫЕ для теста, если сервер еще не подключен
                userShifts = {
                    "2026-05-28": { type: "Д", route: "74", start: "08:00", end: "20:00", loco: "ЭД4М-0450", note: "Явка ст. Раменское" },
                    "2026-05-31": { type: "Н", route: "12У", start: "20:00", end: "08:00", loco: "ЭП2Д-0120", note: "Сдача в депо" }
                };
                document.getElementById("authScreen").style.style.display = "none";
                document.getElementById("appScreen").style.display = "flex";
                renderCalendar();
            }
        }

        function renderCalendar() {
            const grid = document.getElementById("calendarGrid");
            // Удаляем старые дни (оставляя шапку Пн-Вс)
            const weekdays = grid.querySelectorAll('.weekday');
            grid.innerHTML = "";
            weekdays.forEach(w => grid.appendChild(w));

            const year = currentDate.getFullYear();
            const month = currentDate.getMonth();
            
            const monthsRu = ["Январь", "Февраль", "Март", "Апрель", "Май", "Июнь", "Июль", "Август", "Сентябрь", "Октябрь", "Ноябрь", "Декабрь"];
            document.getElementById("monthTitle").innerText = `${monthsRu[month]} ${year}`;

            // Первый день месяца
            const firstDay = new Date(year, month, 1);
            // Корректируем под русский календарь (чтобы Пн был первым)
            let startDay = firstDay.getDay() - 1;
            if (startDay === -1) startDay = 6;

            // Количество дней в месяце
            const totalDays = new Date(year, month + 1, 0).getDate();

            // Пустые дни до начала месяца
            for (let i = 0; i < startDay; i++) {
                const div = document.createElement("div");
                div.className = "day other-month";
                grid.appendChild(div);
            }

            // Рендерим числа
            for (let d = 1; d <= totalDays; d++) {
                const div = document.createElement("div");
                div.className = "day";
                div.innerText = d;

                // Форматируем дату в ГГГГ-ММ-ДД для поиска в базе
                const dateKey = `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`;
                
                if (userShifts[dateKey]) {
                    const shift = userShifts[dateKey];
                    if (shift.type === "Д") div.classList.add("has-day-shift");
                    if (shift.type === "Н") div.classList.add("has-night-shift");
                    
                    div.onclick = () => showDetails(dateKey, d, shift);
                } else {
                    div.onclick = () => showEmpty(d);
                }

                grid.appendChild(div);
            }
        }

        function changeMonth(direction) {
            currentDate.setMonth(currentDate.getMonth() + direction);
            renderCalendar();
        }

        function showDetails(key, day, shift) {
            // Снимаем выделение со всех дней
            document.querySelectorAll('.day').forEach(d => d.classList.remove('selected'));
            event.target.classList.add('selected');

            document.getElementById("detailsDate").innerText = `Смена на ${day} число`;
            document.getElementById("detailsContent").innerHTML = `
                <div class="info-row"><span class="info-label">Тип смены:</span><span class="info-value" style="color:${shift.type==='Д'?'var(--day-shift)':'var(--night-shift)'}">${shift.type==='Д'?'Дневная':'Ночная'}</span></div>
                <div class="info-row"><span class="info-label">Маршрут:</span><span class="info-value">${shift.route}</span></div>
                <div class="info-row"><span class="info-label">Время:</span><span class="info-value">${shift.start} — ${shift.end}</span></div>
                <div class="info-row"><span class="info-label">Состав/Локомотив:</span><span class="info-value">${shift.loco}</span></div>
                <div class="info-row"><span class="info-label">Примечание:</span><span class="info-value">${shift.note || '—'}</span></div>
            `;
        }

        function showEmpty(day) {
            document.querySelectorAll('.day').forEach(d => d.classList.remove('selected'));
            event.target.classList.add('selected');
            document.getElementById("detailsDate").innerText = `${day} число`;
            document.getElementById("detailsContent").innerHTML = `<p style="color: var(--text-muted); text-align: center;">В этот день смен не запланировано.</p>`;
        }
    </script>
